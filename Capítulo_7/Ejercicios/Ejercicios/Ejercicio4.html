var num_cuenta, saldo_inicio, total_articulos, total_creditos, limite_credito, saldo = 0;
num_cuenta = parseInt(window.prompt("Escribe tu número de cuenta", "0"));
saldo_inicio = parseFloat(window.prompt("Escrige tu saldo al inicio del mes","0"));
total_articulos = parseInt(window.prompt("Escribe el total de todos los artículos cargados por el cliente en este mes","0"));
total_creditos = parseInt(window.prompt("Escribe el total de créditos a los que has aplicados a la cuenta del cliente en este mes","0"));
limite_credito = parseInt(window.prompt("Escribe el límite el crédito permitido","0"));

saldo = saldo_inicio + total_articulos - total_creditos;
document.writeln("<h2>Balance</h2>");
document.writeln("<p>");
document.writeln("<strong>Saldo actual: </strong>"+ saldo);
document.writeln("<br><strong>Límite de crédito permitido: </strong>"+ limite_credito);
if ( saldo > limite_credito)
  document.writeln("<br&gr;<b class = 'negar'>Se excedio el límite de su crédito</b>")
else
  document.writeln("<br><b class = 'aceptar'>Aún no ha excedido el límite de su crédito</b>")
document.writeln("</p>");
